// Замер Р1.6б: множества журнала слотами — тот же путь, каким идёт приложение
// (`record_and_project_all`: журнал плюс проекция в базу).
//
// Запуск (из `core/`): скопировать в `sync/tests/r1_6b_bench.rs`,
//   cargo test --release -p plinth-sync --test r1_6b_bench -- --nocapture
// и удалить файл. Для сравнения с `main` — временно положить в ворктри
// `core/sync/src/journal/doc.rs` из `main` (`git show main:<путь> > <путь>`),
// прогнать, вернуть: `git checkout -- core/sync/src`.
//
// Планировщик, 2026-10-27, ПК, release, база в памяти, 10 000 лайков
// (два прогона `main` — диапазон; справа — временная эталонная реализация слотов):
//
//                                        main 45ea51f              слоты
//   через record_and_project_all:
//     10 000 лайков одной правкой        346–388 мс                383 мс
//     один лайк / одно снятие            3,1–3,7 / 2,0 мс          4,4 / 3,0 мс
//     state()                            16–19 мс                  19 мс
//     снимок / после снятия всех         2 434 749 / 2 184 753     2 814 789 / 2 564 793 байт
//   только журнал (record_all):
//     10 000 лайков / снятий одной правкой   20 / 6,2 мс           29 / 10,0 мс
//     один лайк                          2,4 мс                    1,3 мс
//     state()                            2,9 мс                    6,9 мс
//     снимок / после снятия всех         680 033 / 430 116         1 060 033 / 810 154 байт
//   (в снимке «через проекцию» ещё паспорта, которые она заводит; рост размера — только ключи слотов,
//    +38 байт на лайк; стёртые слоты остаются надгробиями с тем же ключом)

#![allow(clippy::unwrap_used)]

mod common;

use std::time::{Duration, Instant};

use common::Scratch;
use plinth_library::db::Database;
use plinth_library::model::{AudioSpec, OnlineSource, OnlineTrack};
use plinth_sync::journal::{Journal, Op, record_and_project_all};
use plinth_types::{DeviceId, Format, ProviderId, Timestamp};

fn track(i: usize) -> OnlineTrack {
    OnlineTrack {
        title: format!("Title {i}"),
        artist: Some(format!("Artist {}", i % 500)),
        album: Some(format!("Album {}", i % 1000)),
        number: Some(1),
        year: None,
        duration: Some(Duration::from_millis(120_000 + i as u64)),
        mbid: None,
        sources: vec![OnlineSource {
            provider: ProviderId::new("archive.org").unwrap(),
            external_id: format!("item{i}/a.mp3"),
            audio: AudioSpec { format: Format::Mp3, bitrate: None, sample_rate_hz: None, bit_depth: None },
        }],
    }
}

#[test]
fn bench() {
    let dir = Scratch::new();
    let db = Database::open_in_memory().unwrap();
    let mut journal = Journal::open(&dir.journal(), DeviceId::new()).unwrap();
    let ids: Vec<_> = (0..10_002).map(|i| db.add_online_track(&track(i), Timestamp::from_millis(1)).unwrap()).collect();

    // 10 000 лайков одной правкой.
    let likes: Vec<Op> = ids[..10_000].iter().map(|t| Op::Like { track: *t }).collect();
    let t = Instant::now();
    record_and_project_all(&mut journal, &db, &likes).unwrap();
    println!("BENCH 10 000 likes in one edit: {:?}", t.elapsed());

    // Одно действие при 10 000 лайков: каждое обращение к множеству собирает суффиксы установок.
    let t = Instant::now();
    record_and_project_all(&mut journal, &db, &[Op::Like { track: ids[10_000] }]).unwrap();
    println!("BENCH one like: {:?}", t.elapsed());
    let t = Instant::now();
    record_and_project_all(&mut journal, &db, &[Op::Unlike { track: ids[10_000] }]).unwrap();
    println!("BENCH one unlike: {:?}", t.elapsed());

    // Чтение состояния и размер журнала.
    let t = Instant::now();
    let state = journal.state();
    println!("BENCH state() with 10 000 likes: {:?}", t.elapsed());
    assert_eq!(state.likes.len(), 10_000);
    println!("BENCH snapshot with 10 000 likes: {} bytes", journal.snapshot().len());

    // Снять все одной правкой; от стёртых слотов в снимке остаются ключи.
    let unlikes: Vec<Op> = ids[..10_000].iter().map(|t| Op::Unlike { track: *t }).collect();
    let t = Instant::now();
    record_and_project_all(&mut journal, &db, &unlikes).unwrap();
    println!("BENCH 10 000 unlikes in one edit: {:?}", t.elapsed());
    println!("BENCH snapshot after the unlikes: {} bytes", journal.snapshot().len());
    assert!(journal.state().likes.is_empty());

    // Только журнал, без проекции и паспортов, которые она заводит: цена самих лайков и их слотов.
    let dir = Scratch::new();
    let mut bare = Journal::open(&dir.journal(), DeviceId::new()).unwrap();
    let t = Instant::now();
    bare.record_all(&likes).unwrap();
    println!("BENCH journal only, 10 000 likes in one edit: {:?}", t.elapsed());
    println!("BENCH journal only, snapshot with 10 000 likes: {} bytes", bare.snapshot().len());
    let t = Instant::now();
    bare.record(&Op::Like { track: ids[10_000] }).unwrap();
    println!("BENCH journal only, one like: {:?}", t.elapsed());
    let t = Instant::now();
    let state = bare.state();
    println!("BENCH journal only, state(): {:?}", t.elapsed());
    assert_eq!(state.likes.len(), 10_001);
    let t = Instant::now();
    bare.record_all(&unlikes).unwrap();
    println!("BENCH journal only, 10 000 unlikes in one edit: {:?}", t.elapsed());
    println!("BENCH journal only, snapshot after the unlikes: {} bytes", bare.snapshot().len());
}
